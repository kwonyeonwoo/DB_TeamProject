# Physical Schema

기준 문서: `docs/source/requirements.md`, `docs/source/logical-schema.md`  
작성 기준: 요구사항 명세서의 업무 의미와 논리 스키마의 릴레이션/키 정의를 우선한다.

## 작성 방식

- MySQL 전용 문법인 백틱, `ENGINE`, `CHARSET`, `COLLATE`, `COMMENT`, `UNSIGNED`, `AUTO_INCREMENT`는 제외했다.
- 논리 스키마에서 `int`로 정의한 식별자는 `INTEGER GENERATED BY DEFAULT AS IDENTITY`로 표현했다.
- 긴 본문 데이터는 일반 SQL 타입인 `CLOB`으로 표현했다.
- 검색 성능용 인덱스는 테이블 생성 뒤 `CREATE INDEX` 문으로 분리했다.
- `notification.comment_content`는 알림 표시용 댓글 내용으로 보관하며 FK 제약으로 선언하지 않는다.

```sql
CREATE TABLE users (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  login_id VARCHAR(50) NOT NULL,
  password VARCHAR(255) NOT NULL,
  name VARCHAR(50) NOT NULL,
  email_address VARCHAR(255) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
  deleted_at TIMESTAMP,
  status VARCHAR(20) DEFAULT 'ACTIVE' NOT NULL,
  role VARCHAR(20),

  PRIMARY KEY(id),
  UNIQUE(login_id),
  CHECK(status IN ('ACTIVE', 'DELETED'))
);

CREATE TABLE post (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  user_id INTEGER NOT NULL,
  title VARCHAR(255) NOT NULL,
  content CLOB,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
  is_updated BOOLEAN DEFAULT FALSE NOT NULL,
  view_count INTEGER DEFAULT 0 NOT NULL,
  is_reported BOOLEAN DEFAULT FALSE NOT NULL,
  main_category VARCHAR(100) NOT NULL,
  sub_category VARCHAR(100) NOT NULL,

  PRIMARY KEY(id),
  CHECK(is_updated IN (TRUE, FALSE)),
  CHECK(view_count >= 0),
  CHECK(is_reported IN (TRUE, FALSE)),
  FOREIGN KEY(user_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE
);

CREATE TABLE likes (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  user_id INTEGER NOT NULL,
  post_id INTEGER NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

  PRIMARY KEY(id),
  UNIQUE(user_id, post_id),
  FOREIGN KEY(user_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE,
  FOREIGN KEY(post_id) REFERENCES post(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

CREATE TABLE comments (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  user_id INTEGER NOT NULL,
  post_id INTEGER NOT NULL,
  parent_comment INTEGER,
  content CLOB NOT NULL,
  is_public BOOLEAN DEFAULT TRUE NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
  is_updated BOOLEAN DEFAULT FALSE NOT NULL,

  PRIMARY KEY(id),
  CHECK(is_public IN (TRUE, FALSE)),
  CHECK(is_updated IN (TRUE, FALSE)),
  FOREIGN KEY(user_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE,
  FOREIGN KEY(post_id) REFERENCES post(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  FOREIGN KEY(parent_comment) REFERENCES comments(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

CREATE TABLE groups (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  group_link VARCHAR(255) NOT NULL,
  name VARCHAR(100) NOT NULL,
  creator_id INTEGER NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

  PRIMARY KEY(id),
  UNIQUE(group_link),
  FOREIGN KEY(creator_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE
);

CREATE TABLE group_members (
  group_id INTEGER NOT NULL,
  user_id INTEGER NOT NULL,
  role VARCHAR(20) NOT NULL,

  PRIMARY KEY(group_id, user_id),
  CHECK(role IN ('LEADER', 'MEMBER')),
  FOREIGN KEY(group_id) REFERENCES groups(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  FOREIGN KEY(user_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE
);

CREATE TABLE schedules (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  user_id INTEGER NOT NULL,
  group_id INTEGER,
  title VARCHAR(255) NOT NULL,
  start_at TIMESTAMP NOT NULL,
  end_at TIMESTAMP NOT NULL,
  description VARCHAR(500),
  type INTEGER NOT NULL,

  PRIMARY KEY(id),
  CHECK(end_at >= start_at),
  FOREIGN KEY(user_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE,
  FOREIGN KEY(group_id) REFERENCES groups(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

CREATE TABLE file (
  id INTEGER NOT NULL,
  file_url VARCHAR(1024) NOT NULL,

  PRIMARY KEY(id, file_url),
  FOREIGN KEY(id) REFERENCES post(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

CREATE TABLE notification (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  is_read BOOLEAN DEFAULT FALSE NOT NULL,
  comment_content CLOB NOT NULL,
  commented_post_id INTEGER NOT NULL,
  commented_user_id INTEGER NOT NULL,
  commented_id INTEGER NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

  PRIMARY KEY(id),
  CHECK(is_read IN (TRUE, FALSE)),
  FOREIGN KEY(commented_post_id) REFERENCES post(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  FOREIGN KEY(commented_user_id) REFERENCES users(id)
    ON DELETE NO ACTION
    ON UPDATE CASCADE,
  FOREIGN KEY(commented_id) REFERENCES comments(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

-- notification.comment_content는 알림에 표시할 댓글 내용을 보관한다.

CREATE INDEX idx_post_user_id ON post(user_id);
CREATE INDEX idx_post_category ON post(main_category, sub_category);
CREATE INDEX idx_post_created_at ON post(created_at);

CREATE INDEX idx_likes_post_id ON likes(post_id);

CREATE INDEX idx_comments_user_id ON comments(user_id);
CREATE INDEX idx_comments_post_id ON comments(post_id);
CREATE INDEX idx_comments_parent_comment ON comments(parent_comment);
CREATE INDEX idx_comments_visibility ON comments(post_id, is_public);

CREATE INDEX idx_groups_creator_id ON groups(creator_id);

CREATE INDEX idx_group_members_user_id ON group_members(user_id);

CREATE INDEX idx_schedules_user_period ON schedules(user_id, start_at, end_at);
CREATE INDEX idx_schedules_group_period ON schedules(group_id, start_at, end_at);
CREATE INDEX idx_schedules_type ON schedules(type);

CREATE INDEX idx_file_post_id ON file(id);

CREATE INDEX idx_notification_receiver ON notification(commented_user_id, is_read, created_at);
CREATE INDEX idx_notification_post_id ON notification(commented_post_id);
CREATE INDEX idx_notification_comment_id ON notification(commented_id);
```

## DBMS별 적용 참고

- MySQL에 실행할 때는 `GENERATED BY DEFAULT AS IDENTITY`를 `AUTO_INCREMENT`로 바꿀 수 있다.
- MySQL에 실행할 때는 `CLOB` 대신 `TEXT`를 사용할 수 있다.
- PostgreSQL, H2, Oracle 계열은 `GENERATED BY DEFAULT AS IDENTITY` 형식을 비교적 그대로 사용할 수 있다.
